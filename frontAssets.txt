    function updateSenderFilter(chat) {

    function loadMessages(messages, highlightText = '') {
        const messagesContainer = document.getElementById('chat-messages');
        messagesContainer.innerHTML = '';

        messages.forEach(msg => {
            const messageElement = document.createElement('div');
            messageElement.className = `message ${msg.outgoing ? 'message-outgoing' : 'message-incoming'}`;

            let messageText = msg.text;
            if (highlightText && msg.text.toLowerCase().includes(highlightText.toLowerCase())) {
                const regex = new RegExp(highlightText, 'gi');
                messageText = msg.text.replace(regex, match => `<span class="highlight">${match}</span>`);
            }

            let messageContent = `
                ${!msg.outgoing ? `<div class="message-sender">${msg.sender}</div>` : ''}
                <div>${messageText}</div>
            `;

            if (msg.file) {
                const fileIcon = msg.file.type === 'pdf' ? '📄' :
                               msg.file.type === 'image' ? '🖼️' : '📎';
                messageContent += `
                    <div class="file-message">
                        <span class="file-icon">${fileIcon}</span>
                        <span class="file-name">${msg.file.name}</span>
                    </div>
                `;
            }

            const messageDate = new Date(msg.time);
            const timeString = messageDate.toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'});

            messageContent += `<div class="message-time">${timeString}</div>`;

            messageElement.innerHTML = messageContent;
            messagesContainer.appendChild(messageElement);
        });

        // Прокручиваем вниз
        messagesContainer.scrollTop = messagesContainer.scrollHeight;
    }

    function searchMessages() {
        if (!currentChatId) return;

        const searchText = document.getElementById('search-messages').value;
        const dateFrom = document.getElementById('date-from').value;
        const dateTo = document.getElementById('date-to').value;
        const sender = document.getElementById('sender-filter').value;

        const chat = chats[currentChatId];
        let filteredMessages = [...chat.messages];

        // Фильтрация по отправителю
        if (sender) {
            filteredMessages = filteredMessages.filter(msg => msg.sender === sender);
        }

        // Фильтрация по дате
        if (dateFrom) {
            const fromDate = new Date(dateFrom);
            filteredMessages = filteredMessages.filter(msg => new Date(msg.time) >= fromDate);
        }

        if (dateTo) {
            const toDate = new Date(dateTo);
            toDate.setDate(toDate.getDate() + 1); // Включаем весь выбранный день
            filteredMessages = filteredMessages.filter(msg => new Date(msg.time) < toDate);
        }

        // Поиск по тексту
        if (searchText) {
            filteredMessages = filteredMessages.filter(msg =>
                msg.text.toLowerCase().includes(searchText.toLowerCase())
            );
        }

        // Показываем результаты
        loadMessages(filteredMessages, searchText);

        // Информация о результатах поиска
        let infoText = '';
        if (searchText || dateFrom || dateTo || sender) {
            infoText = `Найдено сообщений: ${filteredMessages.length}`;
            if (filteredMessages.length === 0) {
                infoText += ' (нет результатов)';
            }
        }
        document.getElementById('search-results-info').textContent = infoText;
    }

    function generateChatReport() {
        if (!currentChatId) return;

        const chat = chats[currentChatId];
        const reportContent = document.getElementById('report-content');

        // Форматируем дату создания отчёта
        const now = new Date();
        const reportDate = now.toLocaleDateString('ru-RU', {
            day: 'numeric',
            month: 'long',
            year: 'numeric',
            hour: '2-digit',
            minute: '2-digit'
        });

        let reportHTML = `
            <h3>${chat.title}</h3>
            <p><strong>Дата создания отчёта:</strong> ${reportDate}</p>
            <p><strong>Участники:</strong> ${chat.participants}</p>
            <p><strong>Всего сообщений:</strong> ${chat.messages.length}</p>
            <hr/>
            <h4>Статистика активности:</h4>
            <ul>
        `;

        // Собираем статистику по участникам
        const participantsStats = {};
        chat.messages.forEach(msg => {
            if (!msg.outgoing) {
                participantsStats[msg.sender] = (participantsStats[msg.sender] || 0) + 1;
            }
        });

        // Преобразуем статистику в массив и сортируем по количеству сообщений
        const sortedParticipants = Object.entries(participantsStats)
            .sort((a, b) => b[1] - a[1]);

        sortedParticipants.forEach(([participant, count]) => {
            reportHTML += `<li>${participant}: ${count} сообщений</li>`;
        });

        reportHTML += `
            </ul>
            <hr/>
            <h4>Последние сообщения:</h4>
        `;

        // Добавляем последние 5 сообщений
        const lastMessages = chat.messages.slice(-5);
        lastMessages.forEach(msg => {
            const messageDate = new Date(msg.time);
            const dateString = messageDate.toLocaleDateString('ru-RU', {
                day: 'numeric',
                month: 'long',
                hour: '2-digit',
                minute: '2-digit'
            });

            reportHTML += `
                <div style="margin-bottom: 1rem; padding: 0.5rem; background: #f3f4f6; border-radius: 8px;">
                    <div><strong>${msg.sender}</strong> (${dateString})</div>
                    <div>${msg.text}</div>
                    ${msg.file ? `<div>📎 Прикреплён файл: ${msg.file.name}</div>` : ''}
                </div>
            `;
        });

        reportContent.innerHTML = reportHTML;

        // Показываем окно отчёта
        document.getElementById('report-window').classList.remove('hidden');
    }

    function applyPreset(preset) {
  const lang = document.getElementById('languageSelect');
  const status = document.getElementById('statusCheckbox');
  const invitable = document.getElementById('invitableCheckbox');

  if (preset === 'default') {
    lang.value = '0';           // Русский
    status.checked = true;
    invitable.checked = true;
  } else if (preset === 'stealth') {
    lang.value = '0';           // Русский
    status.checked = false;
    invitable.checked = false;
  }
  // При "custom" ничего не трогаем
}
